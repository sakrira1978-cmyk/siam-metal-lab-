                .replace(/'/g, "&#039;");
        }

        function startDrawing(e) {
            isDrawing = true;
            const rect = sigCanvas.getBoundingClientRect();

        function initIndexedDB() {
            const request = indexedDB.open('SiamMetalLabVaultDB_v3', 1);
            request.onerror = (event) => console.error('IndexedDB error:', event.target.error);
            request.onsuccess = (event) => {
                db = event.target.result;
                loadVaultFromDB();
            };
            request.onupgradeneeded = (event) => {
                db = event.target.result;
                if (!db.objectStoreNames.contains('vault')) {
                    db.createObjectStore('vault', { keyPath: 'id', autoIncrement: true });
                }
            };
        }

        function loadVaultFromDB() {
            if (!db) {
                renderVault();
                return;
            }
            const transaction = db.transaction(['vault'], 'readonly');
            const store = transaction.objectStore('vault');
            const request = store.getAll();
            request.onsuccess = (event) => {
                const results = event.target.result;
                if (results && results.length > 0) {
                    vaultItems = results;
                } else {
                    vaultItems.forEach(item => {
                        const tx = db.transaction(['vault'], 'readwrite');
                        const copyItem = { ...item };
                        delete copyItem.id;
                        tx.objectStore('vault').add(copyItem);
                    });
                }
                renderVault();
                generateLiveJsonFeed();
            };
        }

        function saveVaultItemToDB(item) {
            if (!db) {
                vaultItems.unshift(item);
                renderVault();
                return;
            }
            const transaction = db.transaction(['vault'], 'readwrite');
            const store = transaction.objectStore('vault');
            const itemToSave = { ...item };
            if (itemToSave.id) {
                delete itemToSave.id;
            }
            const request = store.add(itemToSave);
            request.onsuccess = () => {
                loadVaultFromDB();
            };
            request.onerror = (e) => {
                console.error("Save error:", e);
                vaultItems.unshift(item);
                renderVault();
            };
        }

        function handleBatchUpload(event) {
            const files = event.target.files;
            if (!files || files.length === 0) return;

            const materialTypeElement = document.getElementById('select-metal-type');
            const currentMaterial = materialTypeElement ? materialTypeElement.value : 'ผิวเดิม';

            let count = 0;
            for (let i = 0; i < files.length; i++) {
                const file = files[i];
                const reader = new FileReader();
                reader.onload = function(e) {
                    const newItem = {
                        name: `รูปหล่อหลวงพ่อเดิมกรุพนมรอก (${file.name})`,
                        material: currentMaterial,
                        image: e.target.result
                    };
                    saveVaultItemToDB(newItem);
                    count++;
                    if (count === files.length) {
                        alert(`บันทึกภาพเข้าคลังถาวรสำเร็จจำนวน ${files.length} รูป!`);
                    }
                };
                reader.readAsDataURL(file);
            }
        }

        function importDataBackup(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const importedData = JSON.parse(e.target.result);
                    if (Array.isArray(importedData)) {
                        if (db) {
                            const transaction = db.transaction(['vault'], 'readwrite');
                            const store = transaction.objectStore('vault');
                            store.clear();
                            importedData.forEach(item => store.add(item));
                            transaction.oncomplete = () => {
                                loadVaultFromDB();
                                alert(`นำเข้าข้อมูลและโอนย้ายสำเร็จจำนวน ${importedData.length} รายการ!`);
                            };
                        }
                    } else {
                        alert('รูปแบบไฟล์ข้อมูลไม่ถูกต้อง');
                    }
                } catch (err) {
                    console.error(err);
                    alert('เกิดข้อผิดพลาดในการอ่านไฟล์ JSON');
                }
            };
            reader.readAsText(file);
        }

        function deleteVaultItem(id) {
            if (!confirm('คุณต้องการลบภาพนี้ออกจากคลัง Masterpiece ถาวรใช่หรือไม่?')) return;
            if (!db) {
                vaultItems = vaultItems.filter(item => item.id !== id);
                renderVault();
                generateLiveJsonFeed();
                alert('ลบภาพออกจากคลังเรียบร้อยแล้ว');
                return;
            }
            const transaction = db.transaction(['vault'], 'readwrite');
            const store = transaction.objectStore('vault');
            const request = store.delete(isNaN(id) ? id : Number(id));
            request.onsuccess = () => {
                loadVaultFromDB();
                alert('ลบภาพออกจากคลังเรียบร้อยแล้ว');
            };
            request.onerror = (e) => {
                console.error("Delete error:", e);
                vaultItems = vaultItems.filter(item => item.id !== id);
                renderVault();
                generateLiveJsonFeed();
                alert('ลบภาพออกจากคลังเรียบร้อยแล้ว');
            };
        }
